res.status(201).json({ success: true, user: result.rows[0] }); } catch (error) { res.status(500).json({ error: error.message }); } }); /* ========================================================= ADD PRODUCT ========================================================= */ app.post("/api/products", async (req, res) => { try { const { seller_id, name, description, category, price_ugx, stock_quantity, image_url } = req.body; if ( !seller_id || !name || !category || price_ugx === undefined ) { return res.status(400).json({ error: "Required product information missing." }); } const result = await pool.query(` INSERT INTO products ( seller_id, name, description, category, price_ugx, stock_quantity, image_url ) VALUES ($1,$2,$3,$4,$5,$6,$7) RETURNING * `, [ seller_id, name, description || null, category, price_ugx, stock_quantity || 0, image_url || null ]); } await client.query(` INSERT INTO audit_logs ( event_type, entity_type, entity_id, event_data ) VALUES ( 'ORDER_CREATED', 'ORDER', $1, $2 ) `, [ order.id, JSON.stringify({ order_number: order.order_number, total_ugx: total }) ]); await client.query("COMMIT"); res.status(201).json({ success: true, order }); } catch (error) { await client.query("ROLLBACK"); res.status(500).json({ error: error.message }); } finally { client.release(); } }); /* ========================================================= CREATE PAYMENT REQUEST ========================================================= This creates a pending payment record. Provider integration required." }); } catch (error) { res.status(500).json({ error: error.message }); } }); /* ========================================================= PAYMENT CONFIRMATION ========================================================= This endpoint represents the provider callback. In production, it must verify the provider signature/ credentials before accepting the notification. ========================================================= */ app.post("/api/payments/webhook", async (req, res) => { const client = await pool.connect(); try { const { provider, provider_event_id, provider_transaction_id, payment_id, status } = req.body; if ( !provider || !provider_event_id || !payment_id || !status ) { return res.status(400).json({ error: "Invalid webhook." }); if (existing.rows.length > 0) { await client.query("COMMIT"); return res.json({ success: true, message: "Webhook already processed." }); } await client.query(` INSERT INTO payment_webhook_events ( provider, provider_event_id, payload ) VALUES ($1,$2,$3) `, [ provider, provider_event_id, JSON.stringify(req.body) ]); const paymentResult = await client.query(` SELECT * FROM payments WHERE id = $1 FOR UPDATE `, [payment_id]); if (paymentResult.rows.length === 0) { throw new Error( "Payment not found." ); } const payment = paymentResult.rows[0]; /* In production: verify the provider response first. */ if (status !== "SUCCESSFUL") { await client.query(` UPDATE payments SET status = 'FAILED' WHERE id = $1 `, [payment_id]); await client.query("COMMIT"); return res.json({ success: true, payment_status: "FAILED" }); } /* Prevent processing a successful payment twice. */ if (payment.status === "SUCCESSFUL") { await client.query("COMMIT"); return res.json({ success: true, message: "Payment already successful." }); } /* Mark payment successful. */ await client.query(` UPDATE payments SET status = 'SUCCESSFUL', provider_transaction_id = $1, paid_at = NOW() WHERE id = $2 `, [ provider_transaction_id || null, payment_id ]);
